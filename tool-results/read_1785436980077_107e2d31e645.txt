   328→class TaskViewSet(ModelViewSet):
   329→    """ViewSet para Tarefas com ações customizadas."""
   330→
   331→    permission_classes = [IsAuthenticated]
   332→    filter_backends = [filters.SearchFilter, filters.OrderingFilter]
   333→    search_fields = ['title', 'description', 'category', 'internal_notes']
   334→    ordering_fields = ['due_date', 'priority', 'title', 'status', 'created_at']
   335→
   336→    def get_serializer_class(self):
   337→        if self.action == 'list':
   338→            return TaskListSerializer
   339→        if self.action == 'create':
   340→            return TaskCreateSerializer
   341→        return TaskDetailSerializer
   342→
   343→    def get_queryset(self):
   344→        org = getattr(self.request, 'tenant', None)
   345→        if not org:
   346→            return Task.objects.none()
   347→
   348→        qs = Task.objects.filter(organization=org, is_deleted=False)
   349→
   350→        status_param = self.request.query_params.get('status')
   351→        if status_param:
   352→            qs = qs.filter(status=status_param)
   353→
   354→        client_param = self.request.query_params.get('client')
   355→        if client_param:
   356→            qs = qs.filter(client_id=client_param)
   357→
   358→        department_param = self.request.query_params.get('department')
   359→        if department_param:
   360→            qs = qs.filter(department_id=department_param)
   361→
   362→        responsible_param = self.request.query_params.get('responsible')
   363→        if responsible_param:
   364→            qs = qs.filter(
   365→                assignments__user_id=responsible_param,
   366→                assignments__is_active=True,
   367→            ).distinct()
   368→
   369→        priority_param = self.request.query_params.get('priority')
   370→        if priority_param:
   371→            qs = qs.filter(priority=priority_param)
   372→
   373→        origin_param = self.request.query_params.get('origin')
   374→        if origin_param:
   375→            qs = qs.filter(origin=origin_param)
   376→
   377→        return qs.select_related('client', 'department').prefetch_related(
   378→            'assignments__user', 'checklist_items',
   379→        )
   380→
   381→    def perform_create(self, serializer):
   382→        org = getattr(self.request, 'tenant', None)
   383→        if not org:
   384→            raise permissions.PermissionDenied('Organização não definida.')
   385→
   386→        from apps.radar_tasks.services import TaskService
   387→        dados = serializer.validated_data.copy()
   388→        responsible_id = dados.pop('responsible_id', None)
   389→        checklist_items = dados.pop('checklist_items', None)
   390→
   391→        task = TaskService.create_task_avulsa(org, self.request.user, **dados)
   392→
   393→        if responsible_id:
   394→            TaskService.assign_task(task, responsible_id, self.request.user)
   395→
   396→        if checklist_items:
   397→            for idx, item_data in enumerate(checklist_items):
   398→                titulo = item_data.get('title', '').strip()
   399→                if titulo:
   400→                    ChecklistItem(
   401→                        organization=org,
   402→                        task=task,
   403→                        title=titulo,
   404→                        description=item_data.get('description', ''),
   405→                        is_required=item_data.get('is_required', True),
   406→                        display_order=item_data.get('display_order', idx),
   407→                    ).save()
   408→
   409→        serializer.instance = task
   410→
   411→    def perform_update(self, serializer):
   412→        task = serializer.save()
   413→        from apps.audit.services import AuditService
   414→        AuditService.log(
   415→            action='task.updated',
   416→            target_type='task',
   417→            target_id=task.uuid,
   418→            actor=self.request.user,
   419→            organization=task.organization,
   420→            metadata={'title': task.title},
   421→        )
