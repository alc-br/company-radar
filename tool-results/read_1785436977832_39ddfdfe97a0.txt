    38→class TaskForm(forms.ModelForm):
    39→    """Formulário para criação/edição de tarefas avulsas."""
    40→
    41→    class Meta:
    42→        model = Task
    43→        fields = [
    44→            'title', 'description', 'client', 'department', 'category',
    45→            'priority', 'due_date', 'estimated_duration',
    46→            'is_visible_in_portal', 'portal_instructions', 'team_instructions',
    47→            'internal_notes',

    50→            'title': forms.TextInput(attrs={'class': 'form-control'}),
    51→            'description': forms.Textarea(attrs={'class': 'form-control', 'rows': 4}),
    52→            'client': forms.Select(attrs={'class': 'form-control'}),
    53→            'department': forms.Select(attrs={'class': 'form-control'}),
    54→            'category': forms.TextInput(attrs={'class': 'form-control'}),
    55→            'priority': forms.Select(attrs={'class': 'form-control'}),
    56→            'due_date': forms.DateTimeInput(attrs={'class': 'form-control', 'type': 'datetime-local'}),
    57→            'estimated_duration': forms.NumberInput(attrs={'class': 'form-control', 'min': 0}),
    58→            'is_visible_in_portal': forms.CheckboxInput(attrs={'class': 'form-check-input'}),
    59→            'portal_instructions': forms.Textarea(attrs={'class': 'form-control', 'rows': 3}),
    60→            'team_instructions': forms.Textarea(attrs={'class': 'form-control', 'rows': 3}),
    61→            'internal_notes': forms.Textarea(attrs={'class': 'form-control', 'rows': 3}),
    62→        }
    63→
    64→    def __init__(self, *args, **kwargs):
    65→        org = kwargs.pop('organization', None)
    66→        super().__init__(*args, **kwargs)
    67→        if org:
    68→            self.fields['client'].queryset = ClientCompany.objects.filter(
    69→                organization=org, is_deleted=False,
    70→            ).order_by('razao_social')
    71→            self.fields['department'].queryset = Department.objects.filter(
    72→                organization=org, is_active=True,

   210→@login_required
   211→def task_list(request):
   212→    """Lista de tarefas com filtros, busca e paginação.
   213→
   214→    Suporta visualização em tabela e ações em massa.
   215→    """
   216→    org = _get_org(request)
   217→    if not org:
   218→        return _org_redirect(request)
   219→
   220→    # Parâmetros de filtro
   221→    status_filter = request.GET.get('status')
   222→    client_id = request.GET.get('client')
   223→    department_id = request.GET.get('department')
   224→    responsible_id = request.GET.get('responsible')
   225→    priority_filter = request.GET.get('priority')
   226→    search_term = request.GET.get('q', '')
   227→    due_before = request.GET.get('due_before')
   228→    due_after = request.GET.get('due_after')
   229→    origin_filter = request.GET.get('origin')
   230→    ordering = request.GET.get('ordering', 'due_date')
   231→    page = int(request.GET.get('page', 1))
   232→    page_size = int(request.GET.get('page_size', 20))
   233→
   234→    resultado = TaskSelectors.get_task_list(
   235→        org=org,
   236→        status=status_filter,
   237→        client_id=client_id,
   238→        department_id=department_id,
   239→        responsible_id=responsible_id,
   240→        priority=priority_filter,
   241→        search=search_term,
   242→        due_before=due_before,
   243→        due_after=due_after,
   244→        origin=origin_filter,
   245→        ordering=ordering,
   246→        page=page,
   247→        page_size=page_size,
   248→    )
   249→
   250→    # Dados para filtros no template
   251→    clientes = ClientCompany.objects.filter(
   252→        organization=org, is_deleted=False,
   253→    ).order_by('razao_social')
   254→    departamentos = Department.objects.filter(
   255→        organization=org, is_active=True,
   256→    ).order_by('name')
   257→
   258→    context = {
   259→        'tasks': resultado['results'],
   260→        'page': resultado['page'],
   261→        'page_size': resultado['page_size'],
   262→        'total_pages': resultado['total_pages'],
   263→        'total_count': resultado['total_count'],
   264→        'has_next': resultado['has_next'],
   265→        'has_previous': resultado['has_previous'],
   266→        'status_choices': TaskStatus.choices,
   267→        'priority_choices': TaskPriority.choices,
   268→        'origin_choices': TaskOrigin.choices,
   269→        'clientes': clientes,
   270→        'departamentos': departamentos,
   271→        'filtros': {
   272→            'status': status_filter,
   273→            'q': search_term,
   274→            'client': client_id,
   275→            'department': department_id,
   276→            'responsible': responsible_id,
   277→            'priority': priority_filter,
   278→            'due_before': due_before,
   279→            'due_after': due_after,
   280→            'origin': origin_filter,
   281→            'ordering': ordering,
   282→        },
   283→    }
   284→    return render(request, 'radar_tasks/task_list.html', context)

   288→def task_detail(request, uuid):
   289→    """Detalhe completo de uma tarefa com abas:
   290→
   291→    - Geral (info + atribuições)
   292→    - Checklist
   293→    - Documentos

   305→        return redirect('radar_tasks:task_list')
   306→
   307→    # Dependências

   317→    dependency_form = TaskDependencyForm(organization=org, current_task=task)
   318→    due_date_form = UpdateDueDateForm()
   319→    reopen_cancel_form = ReopenCancelForm()
   320→
   321→    context = {
   322→        'task': task,
   323→        'deps': deps,
   324→        'contagem_status': contagem_status,
   325→        'status_choices': dict(TaskStatus.choices),
   326→        'priority_choices': dict(TaskPriority.choices),
   327→        'comment_form': comment_form,
   328→        'checklist_form': checklist_form,
   329→        'assignment_form': assignment_form,
   330→        'dependency_form': dependency_form,
   331→        'due_date_form': due_date_form,
   332→        'reopen_cancel_form': reopen_cancel_form,
   333→        'active_tab': request.GET.get('tab', 'geral'),
   334→    }
   335→    return render(request, 'radar_tasks/task_detail.html', context)

   338→@login_required
   339→def task_create(request):
   340→    """Formulário para criação de tarefa avulsa."""
   341→    org = _get_org(request)
   342→    if not org:
   343→        return _org_redirect(request)
   344→
   345→    if request.method == 'POST':
   346→        form = TaskForm(request.POST, organization=org)
   347→        if form.is_valid():
   348→            try:
   349→                dados = form.cleaned_data.copy()
   350→                dados.pop('client', None)
   351→                dados.pop('department', None)
   352→                dados['client'] = form.cleaned_data.get('client')
   353→                dados['department'] = form.cleaned_data.get('department')
   354→
   355→                tarefa = TaskService.create_task_avulsa(org, request.user, **dados)
   356→                messages.success(request, f'Tarefa "{tarefa.title}" criada com sucesso!')
   357→                return redirect('radar_tasks:task_detail', uuid=tarefa.uuid)
   358→            except ServiceException as exc:
   359→                messages.error(request, str(exc))
   360→            except Exception as exc:
   361→                messages.error(request, f'Erro ao criar tarefa: {exc}')
   362→    else:
   363→        form = TaskForm(organization=org)
   364→
   365→    context = {'form': form, 'action': 'create'}
   366→    return render(request, 'radar_tasks/task_form.html', context)

   369→@login_required
   370→def task_edit(request, uuid):
   371→    """Formulário para edição de tarefa."""
   372→    org = _get_org(request)
   373→    if not org:
   374→        return _org_redirect(request)
   375→
   376→    task = TaskSelectors.get_task_by_uuid(uuid, org)
   377→    if not task:
   378→        messages.error(request, 'Tarefa não encontrada.')
   379→        return redirect('radar_tasks:task_list')
   380→
   381→    if request.method == 'POST':
   382→        form = TaskForm(request.POST, instance=task, organization=org)
   383→        if form.is_valid():
   384→            try:
   385→                campos_editaveis = [
   386→                    'title', 'description', 'client', 'department',
   387→                    'category', 'priority', 'due_date', 'estimated_duration',
   388→                    'is_visible_in_portal', 'portal_instructions',
   389→                    'team_instructions', 'internal_notes',
   390→                ]
   391→                for campo in campos_editaveis:
   392→                    if campo in form.cleaned_data:
   393→                        setattr(task, campo, form.cleaned_data[campo])
   394→                task.full_clean()
   395→                task.save()
   396→
   397→                from apps.audit.services import AuditService
   398→                AuditService.log(
   399→                    action='task.updated',
   400→                    target_type='task',
   401→                    target_id=task.uuid,
   402→                    actor=request.user,
   403→                    organization=org,
   404→                    metadata={'title': task.title},
   405→                )
   406→
   407→                messages.success(request, f'Tarefa "{task.title}" atualizada com sucesso!')
   408→                return redirect('radar_tasks:task_detail', uuid=task.uuid)
   409→            except Exception as exc:
   410→                messages.error(request, f'Erro ao atualizar: {exc}')
   411→    else:
   412→        form = TaskForm(instance=task, organization=org)
   413→
   414→    context = {'form': form, 'object': task, 'action': 'edit'}
   415→    return render(request, 'radar_tasks/task_form.html', context)

   422→    if not org:
   423→        return _org_redirect(request)

   425→    client_id = request.GET.get('client')
   426→    department_id = request.GET.get('department')
   427→

   429→        org,
   430→        client_id=client_id,
   431→        department_id=department_id,

   436→    for status_val, status_label in TaskStatus.choices:
   437→        colunas[status_val] = {
   438→            'label': status_label,
   439→            'value': status_val,
   440→            'tasks': [],
   441→        }
   442→
   443→    for tarefa in tarefas:
   444→        if tarefa.status in colunas:
   445→            colunas[tarefa.status]['tasks'].append(tarefa)
   446→
   447→    clientes = ClientCompany.objects.filter(
   448→        organization=org, is_deleted=False,
   449→    ).order_by('razao_social')
   450→    departamentos = Department.objects.filter(
   451→        organization=org, is_active=True,
   452→    ).order_by('name')
   453→
   454→    context = {
   455→        'colunas': colunas,
   456→        'clientes': clientes,
   457→        'departamentos': departamentos,
   458→        'filtros': {'client': client_id, 'department': department_id},
   459→    }
   460→    return render(request, 'radar_tasks/task_board.html', context)

   805→    new_status = request.POST.get('new_status')

   811→    try:

   815→        messages.error(request, 'Dados inválidos.')

   817→

   820→        user=request.user,

   827→            request,
