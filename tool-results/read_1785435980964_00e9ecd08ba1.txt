   110→   110→    @staticmethod
   111→   111→    @transaction.atomic
   112→   112→    def complete_task(task, user, evidence=None):
   113→   113→        """Conclui uma tarefa após validações de dependências, checklist e documentos.
   114→   114→
   115→   115→        Parâmetros:
   116→   116→            task: instância de Task
   117→   117→            user: usuário que conclui
   118→   118→            evidence: (opcional) evidência textual ou URL
   119→   119→        """
   120→   120→        if task.status == TaskStatus.CONCLUIDA:
   121→   121→            raise ValidationError(
   122→   122→                message='Esta tarefa já está concluída.',
   123→   123→                errors={'status': ['Tarefa já concluída.']},
   124→   124→            )
   125→   125→
   126→   126→        # 1. Verificar dependências bloqueantes
   127→   127→        deps_bloqueantes = TaskDependency.objects.filter(
   128→   128→            target_task=task,
   129→   129→            organization=task.organization,
   130→   130→        ).select_related('source_task')
   131→   131→
   132→   132→        for dep in deps_bloqueantes:
   133→   133→            if dep.source_task.status != TaskStatus.CONCLUIDA:
   134→   134→                raise ValidationError(
   135→   135→                    message=(
   136→   136→                        'Não é possível concluir. A tarefa '
   137→   137→                        f'"{dep.source_task.title}" ainda não foi concluída.'
   138→   138→                    ),
   139→   139→                    errors={'dependencies': [dep.source_task.title]},
   140→   140→                )
   141→   141→
   142→   142→        # 2. Verificar checklist obrigatório
   143→   143→        itens_obrigatorios_pendentes = ChecklistItem.objects.filter(
   144→   144→            task=task,
   145→   145→            organization=task.organization,
   146→   146→            is_required=True,
   147→   147→            status=ChecklistStatus.PENDENTE,
   148→   148→        )
   149→   149→        if itens_obrigatorios_pendentes.exists():
   150→   150→            nomes = list(itens_obrigatorios_pendentes.values_list('title', flat=True))
   151→   151→            raise ValidationError(
   152→   152→                message=f'Existem {len(nomes)} item(ns) obrigatório(s) pendente(s) no checklist.',
   153→   153→                errors={'checklist': nomes},
   154→   154→            )
   155→   155→
   156→   156→        # 3. Verificar documentos obrigatórios aprovados
   157→   157→        try:
   158→   158→            from apps.radar_documents.models import DocumentRequest, DocumentRequestStatus
   159→   159→            from apps.radar_templates.models import TemplateDocumentRequirement
   160→   160→
   161→   161→            solicitacoes_pendentes = 0
   162→   162→            if task.template_task:
   163→   163→                requisitos = TemplateDocumentRequirement.objects.filter(
   164→   164→                    template_task=task.template_task,
   165→   165→                    organization=task.organization,
   166→   166→                    is_required=True,
   167→   167→                )
   168→   168→                for req in requisitos:
   169→   169→                    pendente = DocumentRequest.objects.filter(
   170→   170→                        task=task,
   171→   171→                        organization=task.organization,
   172→   172→                        template_requirement=req,
   173→   173→                    ).exclude(
   174→   174→                        status__in=[DocumentRequestStatus.APROVADO, DocumentRequestStatus.CANCELADO],
   175→   175→                    ).exists()
   176→   176→                    if pendente:
   177→   177→                        solicitacoes_pendentes += 1
   178→   178→
   179→   179→            if solicitacoes_pendentes > 0:
   180→   180→                raise ValidationError(
   181→   181→                    message=(
   182→   182→                        f'Existem {solicitacoes_pendentes} solicitação(ões) de documento(s) '
   183→   183→                        f'pendente(s) que precisam ser aprovadas.'
   184→   184→                    ),
   185→   185→                    errors={'documents': [f'{solicitacoes_pendentes} documento(s) pendente(s)']},
   186→   186→                )
   187→   187→        except ValidationError:
   188→   188→            raise
   189→   189→        except Exception:
   190→   190→            pass  # radar_documents pode não estar disponível
   191→   191→
   192→   192→        # 4. Registrar conclusão
   193→   193→        task.status = TaskStatus.CONCLUIDA
   194→   194→        task.completed_at = timezone.now()
   195→   195→        task.completed_by = user
   196→   196→        task.save(update_fields=['status', 'completed_at', 'completed_by', 'updated_at'])
   197→   197→
   198→   198→        metadata = {'title': task.title}
   199→   199→        if evidence:
   200→   200→            metadata['evidence'] = evidence
   201→   201→
   202→   202→        AuditService.log(
   203→   203→            action='task.completed',
   204→   204→            target_type='task',
   205→   205→            target_id=task.uuid,
   206→   206→            actor=user,
   207→   207→            organization=task.organization,
   208→   208→            metadata=metadata,
   209→   209→        )
   210→   210→
   211→   211→        # 5. Desbloquear tarefas dependentes
   212→   212→        tarefas_desbloqueadas = 0
   213→   213→        dependentes = TaskDependency.objects.filter(
   214→   214→            source_task=task,
   215→   215→            organization=task.organization,
   216→   216→        ).select_related('target_task')
   217→   217→
   218→   218→        for dep in dependentes:
   219→   219→            alvo = dep.target_task
   220→   220→            # Verifica se TODAS as dependências da tarefa alvo estão concluídas
   221→   221→            todas_concluidas = not TaskDependency.objects.filter(
   222→   222→                target_task=alvo,
   223→   223→                organization=alvo.organization,
   224→   224→            ).exclude(source_task__status=TaskStatus.CONCLUIDA).exists()
   225→   225→
   226→   226→            if todas_concluidas and alvo.status == TaskStatus.BLOQUEADA:
   227→   227→                alvo.status = TaskStatus.A_FAZER
   228→   228→                alvo.save(update_fields=['status', 'updated_at'])
   229→   229→                tarefas_desbloqueadas += 1
   230→   230→
   231→   231→                AuditService.log(
   232→   232→                    action='task.unblocked',
   233→   233→                    target_type='task',
   234→   234→                    target_id=alvo.uuid,
   235→   235→                    actor=user,
   236→   236→                    organization=alvo.organization,
   237→   237→                    metadata={'unblocked_by_task': str(task.uuid)},
   238→   238→                )
   239→   239→
   240→   240→                # Notificar seguidores da tarefa desbloqueada
   241→   241→                _notificar_seguidores_tarefa(
   242→   242→                    alvo,
   243→   243→                    template_code='task_unblocked',
   244→   244→                    context={'task_title': alvo.title, 'unblocked_by': task.title},
   245→   245→                )
   246→   246→
   247→   247→        if tarefas_desbloqueadas > 0:
   248→   248→            logger.info(
   249→   249→                'Tarefa concluída desbloqueou %d tarefa(s): uuid=%s',
   250→   250→                tarefas_desbloqueadas, task.uuid,
   251→   251→            )
   252→   252→
   253→   253→        logger.info('Tarefa concluída: uuid=%s, por=%d', task.uuid, user.id)
   254→   254→        return task

   326→   326→    @staticmethod
   327→   327→    @transaction.atomic
   328→   328→    def change_status(task, user, new_status):
   329→   329→        """Altera o status de uma tarefa com validação de máquina de estados."""
   330→   330→        transicoes = _TRANSICOES_VALIDAS.get(task.status, [])
   331→   331→        if new_status not in transicoes:
   332→   332→            status_atual = task.get_status_display()

   334→   334→                message=f'Transição inválida de "{status_atual}" para o status solicitado.',

   336→   336→            )
   337→   337→
   338→   338→        status_anterior = task.status
   339→   339→        task.status = new_status
   340→   340→        task.save(update_fields=['status', 'updated_at'])
   341→   341→
   342→   342→        AuditService.log(
   343→   343→            action='task.status_changed',
   344→   344→            target_type='task',
   345→   345→            target_id=task.uuid,
   346→   346→            actor=user,
   347→   347→            organization=task.organization,
   348→   348→            metadata={
   349→   349→                'title': task.title,
   350→   350→                'from': status_anterior,
   351→   351→                'to': new_status,
   352→   352→            },
   353→   353→        )
   354→   354→
   355→   355→        logger.info(
   356→   356→            'Status alterado: uuid=%s, %d → %d, por=%d',
   357→   357→            task.uuid, status_anterior, new_status, user.id,

   363→   363→    @staticmethod
   364→   364→    @transaction.atomic
   365→   365→    def update_due_date(task, user, new_date, reason=None):
   366→   366→        """Atualiza o prazo de uma tarefa.
   367→   367→
   368→   368→        Se a tarefa estiver vencida, exige motivo da alteração.
   369→   369→        """
   370→   370→        if task.status in (TaskStatus.CONCLUIDA, TaskStatus.CANCELADA):
   371→   371→            raise ValidationError(
   372→   372→                message='Não é possível alterar o prazo de tarefas concluídas ou canceladas.',
   373→   373→                errors={'status': ['Tarefa inativa.']},
   374→   374→            )
   375→   375→
   376→   376→        if task.is_overdue and (not reason or not reason.strip()):
   377→   377→            raise ValidationError(
   378→   378→                message='Tarefa vencida. Informe o motivo da alteração do prazo.',
   379→   379→                errors={'reason': ['Motivo obrigatório para tarefa vencida.']},
   380→   380→            )
   381→   381→
   382→   382→        data_anterior = task.due_date
   383→   383→        task.due_date = new_date
   384→   384→        task.save(update_fields=['due_date', 'updated_at'])
   385→   385→
   386→   386→        metadata = {
   387→   387→            'title': task.title,
   388→   388→            'previous_due_date': str(data_anterior) if data_anterior else None,
   389→   389→            'new_due_date': str(new_date),
   390→   390→        }
   391→   391→        if reason:
   392→   392→            metadata['reason'] = reason.strip()
   393→   393→
   394→   394→        AuditService.log(
   395→   395→            action='task.due_date_updated',
   396→   396→            target_type='task',
   397→   397→            target_id=task.uuid,
   398→   398→            actor=user,
   399→   399→            organization=task.organization,
   400→   400→            metadata=metadata,
   401→   401→        )
   402→   402→
   403→   403→        logger.info(
   404→   404→            'Prazo atualizado: uuid=%s, %s → %s, por=%d',
   405→   405→            task.uuid, data_anterior, new_date, user.id,
   406→   406→        )
   407→   407→        return task

   411→   411→    @staticmethod

   414→   414→        """Atribui um usuário a uma tarefa.

   425→   425→                resource_type='user',
   426→   426→                resource_id=user_id,

   428→   428→
   429→   429→        atribuicao, created = TaskAssignment.objects.update_or_create(
   430→   430→            task=task,

   444→   444→                    'task_uuid': str(task.uuid),
   445→   445→                    'task_title': task.title,
   446→   446→                    'assigned_user_id': user.id,

   448→   448→                    'role': role,
   449→   449→                },
   450→   450→            )

   452→   452→            # Notificar o usuário atribuído

   456→   456→                    channel='email',

   459→   459→                    context={
   460→   460→                        'task_title': task.title,
   461→   461→                        'assigned_by': assigned_by.get_full_name(),
   462→   462→                        'role': role,
   463→   463→                        'task_uuid': str(task.uuid),
   464→   464→                    },
   465→   465→                )
   466→   466→            except Exception as exc:
   467→   467→                logger.warning('Falha ao notificar atribuição: %s', exc)
   468→   468→
   469→   469→            logger.info(
   470→   470→                'Usuário %d atribuído à tarefa %s como %s',
   471→   471→                user.id, task.uuid, role,
   472→   472→            )
   473→   473→        else:
   474→   474→            logger.info(
   475→   475→                'Atribuição reativada: usuário %d na tarefa %s',
   476→   476→                user.id, task.uuid,
   477→   477→            )
   478→   478→
   479→   479→        return atribuicao

   481→   481→    @staticmethod

   483→   483→    def remove_assignment(task, user_id, removed_by):
   484→   484→        """Remove (desativa) uma atribuição de usuário de uma tarefa."""
   485→   485→        try:

   716→   716→    @transaction.atomic
   717→   717→    def bulk_update_status(task_ids, user, new_status, **kwargs):
   718→   718→        """Atualiza o status de múltiplas tarefas, com auditoria por item.
   719→   719→
   720→   720→        Retorna dicionário com contadores de sucesso e erros.
   721→   721→        """
   722→   722→        from apps.radar_tasks.selectors import TaskSelectors
   723→   723→
   724→   724→        resultados = {'sucesso': [], 'erros': [], 'total_processadas': 0}
   725→   725→
   726→   726→        if not task_ids:
   727→   727→            return resultados
   728→   728→
   729→   729→        tarefas = Task.objects.filter(
   730→   730→            uuid__in=task_ids,
   731→   731→            organization=user.organization if hasattr(user, 'organization') else None,
   732→   732→            is_deleted=False,
   733→   733→        )
   734→   734→
   735→   735→        # Se org foi passado explicitamente
   736→   736→        org = kwargs.get('organization')
   737→   737→        if org:
   738→   738→            tarefas = tarefas.filter(organization=org)
   739→   739→
   740→   740→        for tarefa in tarefas:
   741→   741→            try:
   742→   742→                TaskService.change_status(tarefa, user, new_status)
   743→   743→                resultados['sucesso'].append(str(tarefa.uuid))
   744→   744→            except (ValidationError, ServiceException) as exc:
   745→   745→                resultados['erros'].append({
   746→   746→                    'uuid': str(tarefa.uuid),
   747→   747→                    'erro': str(exc.message),
   748→   748→                })
   749→   749→            resultados['total_processadas'] += 1
   750→   750→
   751→   751→        logger.info(
   752→   752→            'Bulk status update: new_status=%d, sucesso=%d, erros=%d',
   753→   753→            new_status, len(resultados['sucesso']), len(resultados['erros']),
   754→   754→        )
   755→   755→        return resultados

   758→   758→
   759→   759→    @staticmethod
   760→   760→    def get_my_queue(user, org):
   761→   761→        """Retorna tarefas atribuídas ao usuário, ordenadas por:
   762→   762→
   763→   763→        1. Vencidas primeiro
   764→   764→        2. Prazo mais próximo
   765→   765→        3. Prioridade mais alta
   766→   766→        """
   767→   767→        from django.db.models import Case, When, Value, IntegerField
   768→   768→
   769→   769→        tarefas = (
   770→   770→            Task.objects.filter(
   771→   771→                organization=org,
   772→   772→                is_deleted=False,
   773→   773→                assignments__user=user,
   774→   774→                assignments__is_active=True,
   775→   775→            )
   776→   776→            .exclude(status__in=[TaskStatus.CONCLUIDA, TaskStatus.CANCELADA])
   777→   777→            .select_related('client', 'department')
   778→   778→            .prefetch_related('assignments__user', 'checklist_items')
   779→   779→            .annotate(
   780→   780→                _is_overdue=Case(
   781→   781→                    When(is_overdue=True, then=Value(1)),
   782→   782→                    default=Value(0),
   783→   783→                    output_field=IntegerField(),
   784→   784→                ),
   785→   785→            )
   786→   786→            .order_by('-_is_overdue', 'due_date', '-priority')
   787→   787→        )
   788→   788→        return tarefas
