   389→class TemplateApplyRequestSerializer(serializers.Serializer):
   390→    """Payload para aplicar um template a um cliente."""
   391→    client_uuid = serializers.UUIDField(write_only=True)
   392→    base_date = serializers.DateField(write_only=True)
   393→    variables = serializers.JSONField(required=False, default=dict)
   394→    responsible_mapping = serializers.JSONField(required=False, default=dict)
   395→
   396→    def validate_client_uuid(self, value):
   397→        org = self.context.get('organization')
   398→        if not org:
   399→            return value
   400→        from apps.clients.models import ClientCompany
   401→        try:
   402→            return ClientCompany.objects.get(uuid=value, organization=org, is_deleted=False)
   403→        except ClientCompany.DoesNotExist:
   404→            raise serializers.ValidationError('Cliente não encontrado.')

   410→class TemplateViewSet(viewsets.ModelViewSet):
   411→    """ViewSet para Templates Operacionais."""
   412→
   413→    permission_classes = [IsAuthenticated]
   414→    filter_backends = [filters.SearchFilter, filters.OrderingFilter]
   415→    search_fields = ['name', 'code', 'description']
   416→    ordering_fields = ['name', 'created_at', 'use_count', 'current_version']
   417→    lookup_field = 'uuid'
   418→
   419→    def get_serializer_class(self):
   420→        if self.action == 'list':
   421→            return TemplateListSerializer
   422→        if self.action == 'create':
   423→            return TemplateCreateSerializer
   424→        if self.action in ('update', 'partial_update'):
   425→            return TemplateUpdateSerializer
   426→        if self.action in ('publish', 'duplicate', 'apply'):
   427→            return TemplateDetailSerializer
   428→        return TemplateDetailSerializer
   429→
   430→    def get_queryset(self):
   431→        org = getattr(self.request, 'tenant', None)
   432→        if not org:
   433→            return Template.objects.none()
   434→        return Template.objects.filter(
   435→            organization=org, is_deleted=False,
   436→        ).prefetch_related(
   437→            'versions',
   438→        ).annotate(
   439→            _versions_count=Count('versions'),
   440→        )
   441→
   442→    def perform_create(self, serializer):
   443→        org = getattr(self.request, 'tenant', None)
   444→        if not org:
   445→            raise permissions.PermissionDenied('Organização não definida.')
   446→        dados = serializer.validated_data.copy()
   447→        template = TemplateService.create_template(org, self.request.user, **dados)
   448→        serializer.instance = template
   449→
   450→    def perform_update(self, serializer):
   451→        dados = serializer.validated_data.copy()
   452→        dados['_actor'] = self.request.user
   453→        template = TemplateService.update_template(serializer.instance, **dados)
   454→        serializer.instance = template
   455→
   456→    def perform_destroy(self, instance):
   457→        instance.soft_delete()
   458→
   459→    @action(detail=True, methods=['post'])
   460→    def publish(self, request, uuid=None):
   461→        """Publica o rascunho mais recente do template."""
   462→        template = self.get_object()
   463→        draft = template.versions.filter(is_published=False).order_by('-version_number').first()
   464→        if not draft:
   465→            return Response(
   466→                {'error': 'Nenhum rascunho encontrado para publicar.'},
   467→                status=status.HTTP_400_BAD_REQUEST,
   468→            )
   469→        try:
   470→            published = TemplateService.publish_version(draft, request.user)
   471→            out = TemplateVersionSerializer(published)
   472→            return Response(out.data)
   473→        except Exception as exc:
   474→            return Response({'error': str(exc)}, status=status.HTTP_400_BAD_REQUEST)
   475→
   476→    @action(detail=True, methods=['post'])
   477→    def duplicate(self, request, uuid=None):
   478→        """Duplica o template."""
   479→        template = self.get_object()
   480→        new_name = request.data.get('name')
   481→        try:
   482→            new_template = TemplateService.duplicate_template(
   483→                template, request.user, new_name=new_name,
   484→            )
   485→            out = TemplateDetailSerializer(new_template)
   486→            return Response(out.data, status=status.HTTP_201_CREATED)
   487→        except Exception as exc:
   488→            return Response({'error': str(exc)}, status=status.HTTP_400_BAD_REQUEST)
   489→
   490→    @action(detail=True, methods=['post'])
   491→    def apply(self, request, uuid=None):
   492→        """Aplica o template a um cliente."""
   493→        template = self.get_object()
   494→        org = getattr(request, 'tenant', None)
   495→        published = TemplateSelectors.get_published_version(template)
   496→        if not published:
   497→            return Response(
   498→                {'error': 'Nenhuma versão publicada disponível.'},
   499→                status=status.HTTP_400_BAD_REQUEST,
   500→            )
   501→        serializer = TemplateApplyRequestSerializer(
   502→            data=request.data,
   503→            context={'organization': org},
   504→        )
   505→        serializer.is_valid(raise_exception=True)
   506→        try:
   507→            client = serializer.validated_data['client_uuid']
   508→            application = TemplateService.apply_template(
   509→                version=published,
   510→                client=client,
   511→                user=request.user,
   512→                base_date=serializer.validated_data['base_date'],
   513→                variables=serializer.validated_data.get('variables', {}),
   514→                responsible_mapping=serializer.validated_data.get('responsible_mapping', {}),
   515→            )
   516→            out = TemplateApplicationSerializer(application)
   517→            return Response(out.data, status=status.HTTP_201_CREATED)
   518→        except Exception as exc:
   519→            return Response({'error': str(exc)}, status=status.HTTP_400_BAD_REQUEST)
   520→
   521→    @action(detail=False, methods=['get'])
   522→    def categories(self, request):
   523→        """Lista categorias disponíveis."""
   524→        org = getattr(request, 'tenant', None)
   525→        if not org:
   526→            return Response([])
   527→        categories = TemplateSelectors.get_categories(org)
   528→        return Response(categories)

   531→class TemplateVersionViewSet(viewsets.ModelViewSet):
   532→    """ViewSet para versões de template."""
   533→
   534→    permission_classes = [IsAuthenticated]
   535→    filter_backends = [filters.OrderingFilter]
   536→    ordering_fields = ['version_number', 'created_at']
   537→    lookup_field = 'uuid'
   538→
   539→    def get_serializer_class(self):
   540→        if self.action == 'list':
   541→            return TemplateVersionSerializer
   542→        if self.action == 'create':
   543→            return TemplateVersionCreateSerializer
   544→        if self.action == 'tree':
   545→            return TemplateVersionDetailSerializer
   546→        return TemplateVersionDetailSerializer
   547→
   548→    def get_queryset(self):
   549→        org = getattr(self.request, 'tenant', None)
   550→        if not org:
   551→            return TemplateVersion.objects.none()
   552→        return TemplateVersion.objects.filter(
   553→            organization=org,
   554→        ).annotate(
   555→            _stages_count=Count('stages'),
   556→        ).select_related('template', 'published_by')
   557→
   558→    def perform_create(self, serializer):
   559→        org = getattr(self.request, 'tenant', None)
   560→        if not org:
   561→            raise permissions.PermissionDenied('Organização não definida.')
   562→        template_id = self.request.data.get('template')
   563→        if not template_id:
   564→            raise serializers.ValidationError({'template': ['template é obrigatório.']})

   741→class TemplateApplicationViewSet(viewsets.ModelViewSet):
   742→    """ViewSet para aplicações de template."""
   743→
   744→    permission_classes = [IsAuthenticated]
   745→    filter_backends = [filters.SearchFilter, filters.OrderingFilter]
   746→    search_fields = ['client__razao_social', 'client__nome_fantasia', 'version__template__name']
   747→    ordering_fields = ['created_at', 'base_date', 'first_due_date']
   748→    lookup_field = 'uuid'
   749→
   750→    def get_serializer_class(self):
   751→        if self.action == 'apply':
   752→            return TemplateApplyRequestSerializer
   753→        return TemplateApplicationSerializer
   754→
   755→    def get_queryset(self):
   756→        org = getattr(self.request, 'tenant', None)
   757→        if not org:
   758→            return TemplateApplication.objects.none()
   759→        return TemplateApplication.objects.filter(
   760→            organization=org,
   761→        ).select_related(
   762→            'client', 'version__template', 'applied_by',
   763→        )
   764→
   765→    @action(detail=False, methods=['post'])
   766→    def apply(self, request):
   767→        """Aplica um template a um cliente (endpoint independente)."""
   768→        org = getattr(request, 'tenant', None)
   769→        if not org:
   770→            raise permissions.PermissionDenied('Organização não definida.')
   771→        serializer = TemplateApplyRequestSerializer(
   772→            data=request.data,
   773→            context={'organization': org},
   774→        )
   775→        serializer.is_valid(raise_exception=True)
   776→        version_uuid = request.data.get('version_uuid')
   777→        if not version_uuid:
   778→            raise serializers.ValidationError({'version_uuid': ['version_uuid é obrigatório.']})
   779→        version = TemplateSelectors.get_version_by_uuid(str(version_uuid), org)
   780→        if not version or not version.is_published:
   781→            raise serializers.ValidationError({'version_uuid': ['Versão não encontrada ou não publicada.']})
   782→        try:
   783→            client = serializer.validated_data['client_uuid']
   784→            application = TemplateService.apply_template(
   785→                version=version,
   786→                client=client,
   787→                user=request.user,
   788→                base_date=serializer.validated_data['base_date'],
   789→                variables=serializer.validated_data.get('variables', {}),
   790→                responsible_mapping=serializer.validated_data.get('responsible_mapping', {}),
   791→            )
   792→            out = TemplateApplicationSerializer(application)
   793→            return Response(out.data, status=status.HTTP_201_CREATED)
   794→        except Exception as exc:
   795→            return Response({'error': str(exc)}, status=status.HTTP_400_BAD_REQUEST)
